def parse_srt(p: Path):
    raw = p.read_text(encoding='utf-8').replace('\r\n', '\n')
    chunks = re.split(r'\n\s*\n', raw.strip())
    cues=[]
    for chunk in chunks:
        lines=[x.strip() for x in chunk.split('\n') if x.strip()]
        if len(lines)<3: continue
        # srt index then timing then words
        timing=lines[1]
        m=re.match(r'(\d\d:\d\d:\d\d),(\d\d\d)\s+-->', timing)
        if not m: continue
        stamp=m.group(1)
        text=' '.join(lines[2:])
        text=re.sub(r'\s+', ' ', text).strip()
        if text:
            cues.append((stamp,text))
    return cues

def without_timestamp(cues):
    # Preserve every textual cue. Consecutive identical cues, if any, are deduplicated.
    words=[]
    prev=None
    for _, text in cues:
        if text != prev:
            words.append(text)
        prev=text
    return '\n'.join(words)
